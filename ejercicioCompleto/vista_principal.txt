
import 'package:flutter/material.dart';
import 'vista_perfil.dart';

class VistaPrincipal extends StatefulWidget {
  const VistaPrincipal({super.key});

  @override
  State<VistaPrincipal> createState() => _VistaPrincipalState();
}

class _VistaPrincipalState extends State<VistaPrincipal> {

  String colorElegido = 'Ninguno';

  // Abrir una pantalla y esperar un resultado
  Future<void> elegirColor() async {

    final resultado = await Navigator.push<String>(
      context,
      MaterialPageRoute(
        builder: (context) => const SeleccionarColorView(),
      ),
    );

    if (!mounted) return;

    if (resultado != null) {
      setState(() {
        colorElegido = resultado;
      });
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Vista principal'),
        backgroundColor: Colors.deepOrange,
        foregroundColor: Colors.white,
      ),

      body: Padding(
        padding: const EdgeInsets.all(20),

        child: ListView(
          children: [

            const Text(
              'Ejemplos de navegación',
              style: TextStyle(
                fontSize: 24,
                fontWeight: FontWeight.bold,
              ),
            ),

            const SizedBox(height: 20),

            // 1. PUSH SIN PARÁMETROS
            ElevatedButton(
              onPressed: () {
                Navigator.push(
                  context,
                  MaterialPageRoute(
                    builder: (context) => const VistaAyuda(),
                  ),
                );
              },
              child: const Text('1. Abrir ayuda (push)'),
            ),

            const SizedBox(height: 12),

            // 2. PUSH NAMED
            ElevatedButton(
              onPressed: () {
                Navigator.pushNamed(
                  context,
                  '/perfil',
                );
              },
              child: const Text('2. Abrir perfil (pushNamed)'),
            ),

            const SizedBox(height: 12),

            // 3. PUSH CON PARÁMETROS
            ElevatedButton(
              onPressed: () {
                Navigator.push(
                  context,
                  MaterialPageRoute(
                    builder: (context) => const VistaPerfil(
                      nombre: 'Angello',
                      edad: 20,
                    ),
                  ),
                );
              },
              child: const Text('3. Perfil con parámetros'),
            ),

            const SizedBox(height: 12),

            // 4. PUSH NAMED CON ARGUMENTS
            ElevatedButton(
              onPressed: () {
                Navigator.pushNamed(
                  context,
                  '/perfil',
                  arguments: {
                    'nombre': 'María',
                    'edad': 22,
                  },
                );
              },
              child: const Text('4. Perfil con arguments'),
            ),

            const SizedBox(height: 25),

            const Divider(),

            const SizedBox(height: 15),

            Text(
              'Color seleccionado: $colorElegido',
              style: const TextStyle(fontSize: 18),
            ),

            const SizedBox(height: 12),

            // 5. ESPERAR UN RESULTADO
            ElevatedButton(
              onPressed: elegirColor,
              child: const Text('5. Elegir un color (await)'),
            ),
          ],
        ),
      ),
    );
  }
}

// OTRA PANTALLA EN EL MISMO ARCHIVO
class VistaAyuda extends StatelessWidget {
  const VistaAyuda({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Ayuda'),
      ),

      body: Center(
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [

            const Text(
              'Esta pantalla está en vista_principal.dart',
              textAlign: TextAlign.center,
            ),

            const SizedBox(height: 20),

            ElevatedButton(
              onPressed: () {
                Navigator.pop(context);
              },
              child: const Text('Volver con pop'),
            ),

            const SizedBox(height: 12),

            ElevatedButton(
              onPressed: () {
                Navigator.pushReplacement(
                  context,
                  MaterialPageRoute(
                    builder: (context) => const VistaPerfil(
                      nombre: 'Usuario desde ayuda',
                    ),
                  ),
                );
              },
              child: const Text('Reemplazar por perfil'),
            ),
          ],
        ),
      ),
    );
  }
}
